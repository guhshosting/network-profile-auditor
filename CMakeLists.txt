cmake_minimum_required(VERSION 3.24)
project(network-profile-auditor VERSION 1.0.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows with MSVC-compatible toolchain is required")
endif()
option(NPA_ANALYZE "Enable MSVC /analyze" OFF)
option(NPA_ASAN "Enable AddressSanitizer" OFF)
if(NPA_ANALYZE AND NPA_ASAN)
  message(FATAL_ERROR "Use separate analysis and ASan builds")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(npa_core STATIC src/cli.cpp src/decode.cpp src/correlation.cpp src/findings.cpp src/text.cpp src/human.cpp src/json.cpp src/profiles.cpp src/interfaces.cpp)
target_include_directories(npa_core PUBLIC include)
target_compile_definitions(npa_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(npa_core PUBLIC ole32 oleaut32 iphlpapi uuid)
add_executable(network-profile-auditor src/main.cpp src/version.rc)
target_link_libraries(network-profile-auditor PRIVATE npa_core)
target_link_options(network-profile-auditor PRIVATE /MANIFESTUAC:level='asInvoker' /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
include(CTest)
if(BUILD_TESTING)
  add_executable(npa_tests tests/tests.cpp)
  target_link_libraries(npa_tests PRIVATE npa_core)
  add_executable(npa_nonadmin tests/nonadmin.cpp)
  target_link_libraries(npa_nonadmin PRIVATE advapi32)
  add_test(NAME synthetic COMMAND npa_tests)
  add_test(NAME live COMMAND npa_tests --live)
  add_test(NAME cli_help COMMAND network-profile-auditor --help)
  add_test(NAME cli_version COMMAND network-profile-auditor --version)
  add_test(NAME cli_json COMMAND network-profile-auditor --json)
  add_test(NAME cli_all COMMAND network-profile-auditor --all)
  set_tests_properties(live cli_json cli_all PROPERTIES LABELS live TIMEOUT 30)
endif()
foreach(t npa_core network-profile-auditor npa_tests npa_nonadmin)
  if(TARGET ${t})
    target_compile_options(${t} PRIVATE /W4 /WX /permissive- /utf-8 /EHsc)
    if(NPA_ANALYZE)
      target_compile_options(${t} PRIVATE /analyze)
    endif()
    if(NPA_ASAN)
      target_compile_options(${t} PRIVATE /fsanitize=address /Zi)
      target_link_options(${t} PRIVATE /INCREMENTAL:NO)
      if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
        message(FATAL_ERROR "NPA_ASAN validation currently requires MSVC")
      endif()
      get_filename_component(npa_compiler_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
      if(CMAKE_SIZEOF_VOID_P EQUAL 8)
        set(npa_asan_name clang_rt.asan_dynamic-x86_64.dll)
      else()
        set(npa_asan_name clang_rt.asan_dynamic-i386.dll)
      endif()
      find_file(npa_asan_runtime "${npa_asan_name}" PATHS "${npa_compiler_dir}" NO_DEFAULT_PATH REQUIRED)
      if(NOT t STREQUAL "npa_core")
        add_custom_command(TARGET ${t} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${npa_asan_runtime}" "$<TARGET_FILE_DIR:${t}>")
      endif()
    endif()
  endif()
endforeach()
install(TARGETS network-profile-auditor RUNTIME DESTINATION bin)
install(FILES README.md JSON.md VALIDATION.md LICENSE DESTINATION share/network-profile-auditor)
